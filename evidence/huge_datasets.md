  SELECT 45 AS part, file_row_number // 4096 AS block,
         count(*) FILTER (WHERE TRUE AND ServiceName = %(service)s) AS n,
         count(*) FILTER (WHERE TRUE AND ServiceName = %(service)s AND lower(SeverityText) IN ('error', 'fatal')) AS err
  FROM read_parquet('https://public-pme.s3.eu-west-3.amazonaws.com/text_bench/part_045.parquet', file_row_number = true)
